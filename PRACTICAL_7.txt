#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, amount;

    // Number of coin types
    cout << "Enter number of coins: ";
    cin >> n;

    vector<int> coins(n);

    // Enter coin values
    cout << "Enter coin values: ";
    for (int i = 0; i < n; i++) {
        cin >> coins[i];
    }

    // Enter amount
    cout << "Enter amount: ";
    cin >> amount;

    // DP array
    vector<int> dp(amount + 1, amount + 1);

    // Base case
    dp[0] = 0;

    // Dynamic Programming
    for (int i = 1; i <= amount; i++) {

        for (int coin : coins) {

            if (coin <= i) {

                int newValue = dp[i - coin] + 1;

                // Replacing min() function
                if (newValue < dp[i]) {
                    dp[i] = newValue;
                }
            }
        }
    }

    // Check if amount is possible
    if (dp[amount] == amount + 1) {
        cout << "Amount cannot be formed using given coins.";
    }
    else {
        cout << "Minimum number of coins = " << dp[amount];
    }

    return 0;
}
